// Substitua o trecho inicial de carregamento e as funções correspondentes no index.html let database = []; // Carrega os dados direto da API Neon async function loadDataFromApi() { showToast("Sincronizando com banco Neon...", "info"); try { const res = await fetch('/api/dispatches'); if (res.ok) { const data = await res.json(); if (data && data.length > 0) { database = data; localStorage.setItem('paulicomp_expedicao_db', JSON.stringify(database)); } else { // Se o banco estiver vazio, usa a base inicial database = [...initialConsolidatedData]; } } else { throw new Error("Falha na API"); } } catch (e) { console.warn("Offline/Erro API, usando cache local:", e); const saved = localStorage.getItem('paulicomp_expedicao_db'); database = saved ? JSON.parse(saved) : [...initialConsolidatedData]; } populateMonthFilter(); updateDashboard(); updateModalStats(); } // Salva novos registros no Neon via API async function saveRecordsToApi(newRecords, newestMonthImported) { showToast("Enviando dados para o banco na nuvem...", "info"); try { const res = await fetch('/api/dispatches', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(newRecords) }); if (res.ok) { const result = await res.json(); showToast(`${result.inserted} novos registros salvos no Neon!`, "success"); await loadDataFromApi(); if (newestMonthImported) { populateMonthFilter(false, newestMonthImported); updateDashboard(); } } else { throw new Error("Erro na gravação remota"); } } catch (err) { showToast("Erro ao sincronizar com Neon. Dados salvos apenas localmente.", "error"); } }
